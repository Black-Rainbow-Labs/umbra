}

# invoicer

A small library for pricing invoices in whole cents.

## Usage

Prices are integers in cents. Tax rates are in basis points, so 825 means 8.25 percent.

--- src/split.mjs
import { roundCents } from "./money.mjs";

/** Splits a total in cents into parts. The parts add up to the total and the first parts get the extra cents. */
export function splitCents(total, parts) {
  const base = roundCents(total / parts);
  const remainder = total - base * parts;
  return Array.from({ length: parts }, (_, index) => base + (index < remainder ? 1 : 0));
}

=== Add a weekend option to dueDate
--- src/dates.mjs
/** Returns the due date as YYYY-MM-DD, the given number of days after the issue date. */
export function dueDate(issued, days, options = {}) {
  const date = new Date(`${issued}T00:00:00Z`);
  const skipWeekends = options.skipWeekends !== false;
  let remaining = days;
  while (remaining > 0) {
    date.setUTCDate(date.getUTCDate() + 1);
    const weekday = date.getUTCDay();
    if (!skipWeekends || (weekday !== 0 && weekday !== 6)) remaining -= 1;
  }
  return date.toISOString().slice(0, 10);
}

--- README.md
# invoicer

A small library for pricing invoices in whole cents.

## Usage

Prices are integers in cents. Tax rates are in basis points, so 825 means 8.25 percent.
Tax is calculated for each line and rounded to the cent.

## Regions

`rateFor(region)` returns the tax rate for CA, NY, TX and WA. Other regions have no tax.

## Refunds

`refundCents(cents, pct)` returns the part of an amount that is refunded. It never refunds a fraction of a cent.

## Due dates

`dueDate(issued, days)` counts calendar days. `dueDate(issued, days, { skipWeekends: true })` counts Monday to Friday only.

## Late fees

`lateFee(cents, daysLate, bpsPerDay)` charges a daily rate and never charges a fraction of a cent.

const SYMBOLS = { USD: "$", EUR: "€", GBP: "£" };

export function formatMoney(cents, currency = "USD") {
  const sign = cents < 0 ? "-" : "";
  const abs = Math.abs(cents);
  return `${sign}${SYMBOLS[currency] ?? "$"}${Math.floor(abs / 100)}.${String(abs % 100).padStart(2, "0")}`;
}
--- src/invoice.mjs
import { taxFor, rateFor } from "./tax/index.mjs";
import { formatMoney } from "./format/index.mjs";

function quantityOf(item) {
  return Math.max(0, item.qty);
}

function priceOf(item) {
  return item.unitCents;
}

export function lineTotal(item) {
  return quantityOf(item) * priceOf(item);
}

export function subtotal(items) {
  return items.reduce((sum, item) => sum + lineTotal(item), 0);
}

export function invoiceTotal(items, bps) {
  return items.reduce((sum, item) => sum + lineTotal(item) + taxFor(lineTotal(item), bps), 0);
}

export function summary(items, bps, currency = "USD") {
  const sub = subtotal(items);
  const total = invoiceTotal(items, bps);
  return [`Subtotal: ${formatMoney(sub, currency)}`, `Tax: ${formatMoney(total - sub, currency)}`, `Total: ${formatMoney(total, currency)}`].join("\n");
}

export function summaryForRegion(items, region, currency = "USD") {
  return summary(items, rateFor(region), currency);
}
