--- src/invoice.mjs
import { taxFor } from "./tax.mjs";
import { formatMoney } from "./format.mjs";

export function lineTotal(item) {
  return item.qty * item.unitCents;
}

export function subtotal(items) {
  return items.reduce((sum, item) => sum + lineTotal(item), 0);
}

export function invoiceTotal(items, bps) {
  return items.reduce((sum, item) => sum + lineTotal(item) + taxFor(lineTotal(item), bps), 0);
}

export function summary(items, bps) {
  const sub = subtotal(items);
  const total = invoiceTotal(items, bps);
  return [`Subtotal: ${formatMoney(sub)}`, `Tax: ${formatMoney(total - sub)}`, `Total: ${formatMoney(total)}`].join("\n");
}

# invoicer

A small library for pricing invoices in whole cents.

## Usage

Prices are integers in cents. Tax rates are in basis points, so 825 means 8.25 percent.
Tax is calculated for each line and rounded to the cent.

## Regions

`rateFor(region)` returns the tax rate for CA, NY, TX and WA. Other regions have no tax.

## Refunds

`refundCents(cents, pct)` returns the part of an amount that is refunded. It never refunds a fraction of a cent.
=== Document the money helpers
--- src/money.mjs
/** Converts a dollar amount to whole cents. */
export function toCents(amount) {
  return Math.round(amount * 100);
}

/** Converts whole cents to a dollar amount with two decimals. */
export function fromCents(cents) {
  return Number((cents / 100).toFixed(2));
}

/** Drops the fraction of a cent. Rounds toward zero, so 2.9 becomes 2 and -2.9 becomes -2. */
export function roundCents(value) {
  return Math.trunc(value);
}
